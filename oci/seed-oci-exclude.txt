# Packages installed by cloth-layer tiers (lint, thread, yarn, fabric,
# tapestry, loom, jacquard) on top of their app-tier parent. Historically
# this file listed "packages excluded from the OCI seed"; post-canopy
# migration (2026-04-21) it's the install list every cloth Containerfile
# feeds to `apt-get install`. Grouped: init/systemd, SSH daemon, D-Bus
# + polkit, cloud-init, auto-updates, kernel/device/security modules.

# Init system — these pull in systemd as a dependency
init
systemd
systemd-resolved
systemd-sysv
systemd-timesyncd
libpam-systemd
libsystemd-shared
udev

# SSH daemon — canopy purges the whole init-family + openssh together
# (deb-systemd-helper in init-system-helpers is needed by openssh's
# postinst to enable ssh.service). Install here at lint where systemd
# + init-system-helpers also land in the same apt transaction, so
# openssh-server's postinst runs with a working environment and
# enables ssh.service naturally. Keeping openssh off seed means app
# tiers (fiber/sheet/page/tome/press/gutenberg) stay sshd-free
# process containers, matching canopy's design intent.
openssh-server
openssh-sftp-server
ssh-import-id

# D-Bus system bus — no init system to start it, nothing uses it
dbus
dbus-bin
dbus-daemon
dbus-session-bus-common
dbus-system-bus-common

# Authorization — needs D-Bus
polkitd

# Cloud metadata — restored intentionally for first-boot config support
# (kento Phase 3.5 NoCloud delivery, ConfigDrive on Proxmox/OpenStack,
# user-data injection by other consumers). Datasource list pinned in
# /etc/cloud/cloud.cfg.d/99-droste.cfg via oci/cloud-init-droste.cfg.
cloud-init
cloud-guest-utils
cloud-utils

# Automatic updates — container images are immutable
unattended-upgrades

# Kernel module management — no kernel access in OCI
kmod

# Device mapper — needs kernel device mapper
dmsetup

# AppArmor — needs kernel AppArmor support
apparmor
